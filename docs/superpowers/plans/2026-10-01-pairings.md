# Pairings in Snow Crew Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move the Crew Board (board, warnings, callouts, gear log) into Snow Crew so Matt places and posts crews there and each crew member sees their own route first.

**Architecture:** The Crew Board's pure modules (`board`, `warnings`, `history`, `gear`, `time`) move into `Snow-App-repo/lib` with their tests; `time` changes from a noon cutover to Matt's 9 AM day start and gains shift ids. The backend gets four tabs (Moves, Callouts, Gear, Posts), admin-only write actions, and builds the crew's posted copy itself from an allow-list, using byte-identical copies of `board`, `time` and `history`. The phone gains Board, Log and Tonight tabs.

**Tech Stack:** Plain ES5-style browser JS (no build), Google Apps Script V8 + clasp, Node `node:test`, Playwright over `file://`.

**Spec:** `docs/superpowers/specs/2026-10-01-pairings-design.md`

## Global Constraints

- Snow-App-repo is PUBLIC: no tracked data JSON (`test/no-data.test.js`), fixtures are inline JS with obviously fake names (Alex Test, Sam Sample…).
- Day shift starts **9 AM**; anything before 9 AM belongs to the night before. Nights are named by their evening's date. No night-start hour exists.
- Shift ids are exactly `night-YYYY-MM-DD` / `day-YYYY-MM-DD`; labels `Night of 10/1` / `Day of 10/2`.
- Shift choices: before 9 AM `[night-<yesterday>, day-<today>, night-<today>]`; from 9 AM `[day-<today>, night-<today>]`; **none pre-picked**.
- Moves, Gear and Posts are **append-only**: no edit or archive action exists for them.
- The crew copy (Posts row) holds only: `posted_at, shift, routes[{id, name, sites[{id,name}], lead, members}], people{id:{name, phone, photo_thumb, shifts{night,day}}}`.
- Warn, never block. `null` never fires or clears a warning.
- `PUBLIC_CREW_FIELDS` stays `['id','name','phone','photo_thumb','is_lead']`.
- Every phone change bumps `sw.js` `CACHE_VERSION`; backend VERSION becomes `pairings-1` with `EXPECTED_BACKEND` in the same release.
- Backend `board.js`, `time.js`, `history.js` are byte-identical to the phone copies (test-enforced), and `.clasp.json` pushes `time.js`, `board.js` before `history.js`.

## Review Focus

1. **Post tapped while a move is still saving** → the post must include that move: Post waits (disabled) until no write is in flight. Test in Task 5.
2. **Archived worker or archived route still in Moves** → never on the board and never in a post. Test in Task 2 (post) and Task 4 (board).
3. **Crew member whose roster row isn't linked to their sign-in** (`me.crew_id` null) → sees "You're not on a route this shift" plus the whole board, never an error. Test in Task 6.
4. **Morning post for tonight** (posted 8 AM 10/1 for `night-2026-10-01`) → shown, not stale. Test in Task 1 (`isStale`) and Task 6.
5. **Site with no `needs_clearance`** → no clearance warning for anyone on that route. Covered by moved warnings tests; asserted again through the site adapter in Task 4.

```js
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.CrewTime = factory();
})(this, function () {
  'use strict';
  // Matt, 10/1/26: day shift starts 9 AM. Anything before belongs to the night
  // before. Nights start when the weather says (6 PM to 3 AM), so there is no
  // night-start hour, and nights are named by their evening's date.
  var SHIFT_CUTOVER_HOUR = 9;

  function pad(n) { return (n < 10 ? '0' : '') + n; }
  function localIso(d) {
    var off = -d.getTimezoneOffset(), sign = off >= 0 ? '+' : '-', ms = d.getMilliseconds();
    off = Math.abs(off);
    return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate()) +
      'T' + pad(d.getHours()) + ':' + pad(d.getMinutes()) + ':' + pad(d.getSeconds()) +
      '.' + (ms < 100 ? '0' : '') + pad(ms) + sign + pad(Math.floor(off / 60)) + ':' + pad(off % 60);
  }
  function addDays(ymd, n) {
    var p = ymd.split('-').map(Number);
    return new Date(Date.UTC(p[0], p[1] - 1, p[2] + n)).toISOString().slice(0, 10);
  }
  // Reads the wall clock straight out of the string: no time-zone dependence.
  function shiftDate(iso) {
    var day = iso.slice(0, 10);
    return Number(iso.slice(11, 13)) >= SHIFT_CUTOVER_HOUR ? day : addDays(day, -1);
  }
  function dayNumber(ymd) { var p = ymd.split('-').map(Number); return Date.UTC(p[0], p[1] - 1, p[2]) / 86400000; }
  function daysBetween(a, b) { return dayNumber(b) - dayNumber(a); }

  function shiftId(kind, date) { return kind + '-' + date; }
  function parseShift(id) {
    var m = /^(night|day)-(\d{4}-\d{2}-\d{2})$/.exec(String(id || ''));
    return m ? { kind: m[1], date: m[2] } : null;
  }
  function shiftLabel(id) {
    var s = parseShift(id); if (!s) return '';
    return (s.kind === 'night' ? 'Night of ' : 'Day of ') + Number(s.date.slice(5, 7)) + '/' + Number(s.date.slice(8, 10));
  }
  // The shifts around now, none pre-picked (the weather sets the night).
  function shiftChoices(iso) {
    var today = iso.slice(0, 10);
    if (Number(iso.slice(11, 13)) < SHIFT_CUTOVER_HOUR) return [shiftId('night', addDays(today, -1)), shiftId('day', today), shiftId('night', today)];
    return [shiftId('day', today), shiftId('night', today)];
  }
  function shiftKey(id) { var s = parseShift(id); return s ? s.date + (s.kind === 'day' ? '1' : '2') : ''; }
  // A post is stale when it is older than every shift on offer now.
  function isStale(postShiftId, iso) { return !parseShift(postShiftId) || shiftKey(postShiftId) < shiftKey(shiftChoices(iso)[0]); }

  return { SHIFT_CUTOVER_HOUR: SHIFT_CUTOVER_HOUR, localIso: localIso, shiftDate: shiftDate, daysBetween: daysBetween,
    shiftId: shiftId, parseShift: parseShift, shiftLabel: shiftLabel, shiftChoices: shiftChoices, shiftKey: shiftKey, isStale: isStale };
});
```

```js
  // "2 nights, 1 day; N1 x2", from callouts. Actually went out, not just placed.
  function shiftsWorked(callouts, workerId) {
    var out = { night: 0, day: 0, byRoute: {} };
    (callouts || []).forEach(function (c) {
      var s = T.parseShift(c.shift); if (!s) return;
      Object.keys(c.roster || {}).forEach(function (r) {
        if (lists(c.roster[r], workerId)) { out[s.kind]++; out.byRoute[r] = (out.byRoute[r] || 0) + 1; }
      });
    });
    return out;
  }

  // Default route for a gear entry: that shift's callout first, then the board as of its date.
  function routeOn(callouts, moves, workerId, shiftId) {
    var c = (callouts || []).filter(function (x) { return x.shift === shiftId; })[0];
    if (c) {
      var hit = Object.keys(c.roster || {}).filter(function (r) { return lists(c.roster[r], workerId); })[0];
      if (hit) return hit;
    }
    var s = T.parseShift(shiftId), route = null;
    if (!s) return null;
    B.sortMoves(moves).forEach(function (m) { if (m.worker === workerId && T.shiftDate(m.at) <= s.date) route = m.to_route; });
    return route;
  }
```

```js
var SnowPairing = (function () {
  'use strict';
  function routesForWarnings(routes, sites) {
    var byId = {}; (sites || []).forEach(function (s) { byId[s.id] = s; });
    return (routes || []).filter(function (r) { return !r.archived; }).map(function (r) {
      return { id: r.id, name: r.name, sites: (r.site_ids || []).filter(function (id) { return byId[id] && !byId[id].archived; })
        .map(function (id) { return { name: byId[id].name, needs_clearance: byId[id].needs_clearance || null }; }) };
    });
  }
  function changedSincePost(board, post) {
    if (!post) return true;
    var posted = {}; (post.routes || []).forEach(function (r) { posted[r.id] = r; });
    return Object.keys(board.routes).some(function (id) {
      var a = board.routes[id], b = posted[id] || { lead: null, members: [] };
      return a.lead !== b.lead || a.members.join() !== (b.members || []).join();
    });
  }
  return { routesForWarnings: routesForWarnings, changedSincePost: changedSincePost };
})();
```
